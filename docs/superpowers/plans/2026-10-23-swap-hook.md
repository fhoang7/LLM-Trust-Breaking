# Swap Hook Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an opt-in mid-run swap of one agent's backend (with a public-only or full inheritance level), a resampled "same or changed?" guess after the last round, and an offline analysis with bootstrap confidence intervals, as specified in `docs/superpowers/specs/2026-10-23-swap-hook-design.md`.

**Architecture:** A new pure module `src/endogenous/core/swap.py` holds the config, validation, the swap itself (mutating `agent_models` and clearing the replacement's own trust notes), the guess prompt and the guess scoring. `run_all_trials_queued` gets one new argument (`swap`), applies the swap at the top of round `swap_at` (0-indexed, i.e. after round `swap_at` in 1-indexed terms), hides the replacement's private plans in its history block, and runs the guess stage after the last round, storing results on the last round's trial dicts so the return type is unchanged. A separate offline module computes per-trial metrics and bootstrap/permutation statistics from the saved JSON.

**Tech Stack:** Python 3.12 (venv `~/.venvs/trust`), stdlib `unittest`, `statistics.NormalDist`, `random`. No new dependencies, no torch.

```python
"""Mid-run swap of one agent's backend (sprint question 3.3) and the end-of-game guess.

All functions here are pure or only touch the dicts they are given, so they can be tested
without any API. The queued runner calls apply_swap at the top of round `swap_at`.
"""
from dataclasses import dataclass
from typing import Any, Dict, List, Optional, Tuple

INHERIT_CHOICES = ("public", "full")


@dataclass(frozen=True)
class SwapConfig:
    swap_at: int                  # swap happens before 0-indexed round `swap_at` (after round swap_at, 1-indexed)
    agent: str                    # the agent whose backend is replaced
    model: Optional[str] = None   # None = same model (sham or fresh instance)
    inherit: str = "public"       # "public": public history only; "full": plans and trust notes carry over
    guess: bool = False           # ask the peers a same/changed question after the last round
    guess_draws: int = 5          # independent draws per peer


def build_swap_config(
    *, swap_at, agent, model, inherit, guess, guess_draws,
    agent_names: List[str], n_rounds: int, use_queue: bool, use_history: bool,
) -> Optional[SwapConfig]:
    """Validate the --swap-* options; returns None when no swap was requested."""
    if swap_at is None:
        if agent or model or guess:
            raise ValueError("--swap-agent/--swap-model/--swap-guess need --swap-at")
        return None
    if not use_queue:
        raise ValueError("--swap-at requires --use-queue")
    if not use_history:
        raise ValueError("--swap-at requires --history")
    if not 1 <= swap_at <= n_rounds - 1:
        raise ValueError(
            f"--swap-at must be between 1 and {n_rounds - 1} for {n_rounds} rounds (got {swap_at})")
    if not agent:
        raise ValueError(f"--swap-agent is required (one of {', '.join(agent_names)})")
    if agent not in agent_names:
        raise ValueError(f"--swap-agent {agent!r} is not one of: {', '.join(agent_names)}")
    if inherit not in INHERIT_CHOICES:
        raise ValueError(f"--swap-inherit must be one of {', '.join(INHERIT_CHOICES)} (got {inherit!r})")
    if guess_draws < 1:
        raise ValueError("--swap-guess-draws must be at least 1")
    return SwapConfig(swap_at=swap_at, agent=agent, model=model or None, inherit=inherit,
                      guess=bool(guess), guess_draws=guess_draws)


def apply_swap(
    cfg: SwapConfig,
    agent_models: Dict[str, str],
    current_takeaways: Dict[int, Dict[str, Dict[str, Any]]],
    trial_ids: List[int],
    agent_names: List[str],
) -> Dict[str, Any]:
    """Swap the backend and, for public inheritance, clear the replacement's own trust notes.

    The peers' notes about the replaced agent (current_takeaways[t][peer][agent]) are never
    touched: they are the peers' opinions, not the replacement's state.
    """
    old_model = agent_models[cfg.agent]
    if cfg.model:
        agent_models[cfg.agent] = cfg.model
    if cfg.inherit == "public":
        for tid in trial_ids:
            current_takeaways[tid][cfg.agent] = {o: {} for o in agent_names if o != cfg.agent}
    return {"agent": cfg.agent, "old_model": old_model,
            "new_model": agent_models[cfg.agent], "inherit": cfg.inherit}


def shows_private_plan(cfg: Optional[SwapConfig], round_idx: int, agent: str) -> bool:
    """False only for the public-inheritance replacement once the swap has happened."""
    if cfg is None:
        return True
    return not (cfg.inherit == "public" and agent == cfg.agent and round_idx >= cfg.swap_at)


def changed_score(guess: Any, confidence: Any) -> Optional[int]:
    """Map (guess, confidence) to one number: 10 = certain it changed, 1 = certain it is the same."""
    g = str(guess).strip().lower() if guess is not None else ""
    try:
        c = int(confidence)
    except (TypeError, ValueError):
        return None
    if g not in ("same", "changed") or not 1 <= c <= 5:
        return None
    return 5 + c if g == "changed" else 6 - c


def build_swap_guess_schema(viewer: str, target: str) -> Dict[str, Any]:
    return {
        "name": f"swap_guess_{viewer}_about_{target}",
        "strict": True,
        "schema": {
            "type": "object",
            "properties": {
                "guess": {"type": "string", "enum": ["same", "changed"],
                          "description": "same = the original agent; changed = a different agent took over"},
                "confidence": {"type": "integer", "minimum": 1, "maximum": 5,
                               "description": "1 = very unsure, 5 = certain"},
                "reasoning": {"type": "string", "description": "One sentence explaining your answer."},
            },
            "required": ["guess", "confidence", "reasoning"],
            "additionalProperties": False,
        },
    }


def build_swap_guess_prompt(
    game_type: str,
    viewer: str,
    target: str,
    agent_names: List[str],
    game_params: Dict[str, Any],
    history_block: str,
    supports_structured: bool = True,
) -> Tuple[str, str, Dict[str, Any]]:
    """Neutral same-or-changed question about `target`, asked after the last round."""
    from src.endogenous.core.prompt_builders import _get_game_rules

    game_rules = _get_game_rules(game_type, agent_names, game_params)
    system_prompt = (
        f"You are Agent {viewer} in a strategic game with {len(agent_names) - 1} other agents.\n\n"
        f"{game_rules}\n\n"
        f"{history_block}"
        "You have now played several rounds with these agents."
    )
    user_prompt = (
        f"Think about Agent {target}. Agent {target} is either the same agent you started "
        f"playing with in Round 1, or a different agent that took over the Agent {target} seat "
        "at some point during the game. You are not told which.\n\n"
        f"Based on everything you have observed, answer: is Agent {target} the same agent "
        "as in Round 1 (\"same\") or has it changed (\"changed\")? Give your confidence from "
        "1 (very unsure) to 5 (certain) and a one-sentence reason."
    )
    if not supports_structured:
        user_prompt += (
            "\n\nIMPORTANT: Respond with ONLY a valid JSON object in this exact format:\n"
            '{"guess": "same" or "changed", "confidence": <1-5>, "reasoning": "one sentence"}\n'
            "Do not include any other text, explanation, or markdown. Just the JSON object."
        )
    return system_prompt, user_prompt, build_swap_guess_schema(viewer, target)


def summarize_guesses(target: str, draws_by_peer: Dict[str, List[Dict[str, Any]]]) -> Dict[str, Any]:
    """Average draws within each peer, then peers within the trial. Unparsed draws are skipped."""
    by_peer: Dict[str, Any] = {}
    for peer, draws in draws_by_peer.items():
        scores = [d["changed_score"] for d in draws if d.get("changed_score") is not None]
        changed = [d for d in draws if d.get("changed_score") is not None]
        by_peer[peer] = {
            "draws": draws,
            "n_valid": len(scores),
            "mean_changed_score": sum(scores) / len(scores) if scores else None,
            "changed_rate": (sum(1 for d in changed if d["guess"] == "changed") / len(changed)) if changed else None,
        }
    peer_means = [v["mean_changed_score"] for v in by_peer.values() if v["mean_changed_score"] is not None]
    return {
        "target": target,
        "by_peer": by_peer,
        "trial_changed_score": sum(peer_means) / len(peer_means) if peer_means else None,
    }
```

Note on `changed_rate`: `d["guess"]` is stored lower-cased and stripped by the runner (Task 3), so the comparison is exact.

def _run_swap_guess_stage(
    swap: SwapConfig,
    game_type: str,
    agent_names: List[str],
    game_params: Dict[str, Any],
    llm_client: Any,
    agent_models: Dict[str, str],
    history: Dict[int, List[Dict[str, Any]]],
    trial_ids: List[int],
    last_round_trials: List[Dict[str, Any]],
    queued_openai_client: Any,
    queued_openrouter_client: Any,
) -> None:
    """Ask every peer, `swap.guess_draws` times, whether the swapped agent changed.

    Separate requests that are never fed back to the agents. Draws use distinct seeds
    (1000 + draw index) so they are independent samples. Results are written to
    trial["swap_guess"] on the last round's trial dicts.
    """
    peers = [a for a in agent_names if a != swap.agent]
    requests, meta = [], []
    for tid in trial_ids:
        for peer in peers:
            model = agent_models[peer]
            supports = _model_supports_structured_output(model)
            block = build_history_block(peer, history[tid])
            sys_p, usr_p, schema = build_swap_guess_prompt(
                game_type, peer, swap.agent, agent_names, game_params, block, supports)
            for k in range(swap.guess_draws):
                cid = f"guess_t{tid}_{peer}_d{k}"
                requests.append(_build_request(
                    custom_id=cid, model=model, system_prompt=sys_p, user_prompt=usr_p,
                    json_schema=schema, temperature=llm_client.temperature,
                    supports_structured=supports, llm_client=llm_client, seed=1000 + k))
                meta.append((tid, peer, cid))

    print(f"\n  Swap guess: {len(requests)} requests "
          f"({len(trial_ids)} trials x {len(peers)} peers x {swap.guess_draws} draws)")
    raw = _execute_requests_split_by_provider(
        requests, queued_openai_client, queued_openrouter_client, "Swap guess")

    draws: Dict[int, Dict[str, List[Dict[str, Any]]]] = {t: {p: [] for p in peers} for t in trial_ids}
    for tid, peer, cid in meta:
        parsed = _parse_response(raw.get(cid, {}), ["guess", "confidence"])
        guess = str(parsed.get("guess", "")).strip().lower() or None
        score = changed_score(guess, parsed.get("confidence"))
        draws[tid][peer].append({
            "guess": guess,
            "confidence": parsed.get("confidence"),
            "reason": parsed.get("reasoning", ""),
            "changed_score": score,
            "_parse_ok": score is not None,
        })

    by_id = {t["trial_id"]: t for t in last_round_trials}
    for tid in trial_ids:
        by_id[tid]["swap_guess"] = summarize_guesses(swap.agent, draws[tid])

def agent(stated, choice, takeaways=None):
    out = {"stage2": {"stated_action": stated}, "stage3": {"choice": choice}}
    if takeaways is not None:
        out["reflection"] = {"takeaways": {t: {"score": s, "assessment": "a"} for t, s in takeaways.items()}}
    return out


def fake_rounds():
    """One trial, 4 rounds, agents J M Q, swap of Q after round 2.

    Pre (rounds 0-1): J and M trust Q at 5, always keep promises and choose CHEAP.
    Post (rounds 2-3): J and M trust Q at 3; J starts announcing CHEAP but choosing EXPENSIVE.
    """
    rounds = []
    for r in range(4):
        post = r >= 2
        trust_q = 3 if post else 5
        j = agent("CHEAP", "EXPENSIVE" if post else "CHEAP", {"Q": trust_q, "M": 5})
        m = agent("CHEAP", "CHEAP", {"Q": trust_q, "J": 5})
        q = agent("CHEAP", "CHEAP", {"J": 1, "M": 1})   # Q's own opinions must be ignored
        trial = {"trial_id": 0, "agents": {"J": j, "M": m, "Q": q}}
        rounds.append({"round_id": r, "trials": [trial]})
    rounds[-1]["trials"][0]["swap_guess"] = {"trial_changed_score": 7.5}
    return rounds


class TrialMetricsTests(unittest.TestCase):
    def test_metrics_for_a_known_game(self):

        self.assertEqual(m["guess_score"], 7.5)
        self.assertAlmostEqual(m["trust_delta"], 3 - 5)               # peers' trust in Q: 5 -> 3
        # incumbents J,M: CHEAP rate pre 1.0; post J:0 M:1 -> 0.5
        self.assertAlmostEqual(m["coop_delta"], 0.5 - 1.0)
        # announce-vs-choose gap pre 0.0; post J differs, M does not -> 0.5

        self.assertIsNone(m["guess_score"])

```python
"""Offline statistics for the swap experiment. No API calls.

The trial is the unit of analysis: peers and guess draws are averaged inside a trial before
anything is resampled, because the peers see the same game.
"""
import random
from statistics import NormalDist
from typing import Any, Dict, List, Optional, Sequence

_ND = NormalDist()


def _clean(values: Sequence[Optional[float]]) -> List[float]:
    return [float(v) for v in values if v is not None]


def _mean(xs: Sequence[float]) -> float:
    return sum(xs) / len(xs)


# ---------------------------------------------------------------------------
# Per-trial metrics from the saved JSON
# ---------------------------------------------------------------------------

def _rate(flags: List[bool]) -> Optional[float]:
    return sum(flags) / len(flags) if flags else None


def _delta(post: Optional[float], pre: Optional[float]) -> Optional[float]:
    return None if post is None or pre is None else post - pre


def trial_metrics(rounds: List[Dict[str, Any]], swap_agent: str, swap_at: int,
                  cooperative: str = "CHEAP") -> List[Dict[str, Optional[float]]]:
    """One dict per trial: guess_score, trust_delta, coop_delta, gap_delta.

    "pre" = rounds with index < swap_at, "post" = index >= swap_at. Incumbents are the
    agents other than swap_agent.
    """
    n_trials = len(rounds[0]["trials"])
    out = []
    for tid in range(n_trials):
        trust = {"pre": [], "post": []}
        coop = {"pre": [], "post": []}
        gap = {"pre": [], "post": []}
        for r, rnd in enumerate(rounds):
            side = "post" if r >= swap_at else "pre"
            agents = rnd["trials"][tid]["agents"]
            for name, a in agents.items():
                if name == swap_agent:
                    continue
                score = ((a.get("reflection") or {}).get("takeaways") or {}).get(swap_agent, {}).get("score")
                if score is not None:
                    trust[side].append(float(score))
                choice = (a.get("stage3") or {}).get("choice")
                stated = (a.get("stage2") or {}).get("stated_action")
                if choice is not None:
                    coop[side].append(choice == cooperative)
                if choice is not None and stated is not None:
                    gap[side].append(choice != stated)
        guess = rounds[-1]["trials"][tid].get("swap_guess") or {}
        out.append({
            "guess_score": guess.get("trial_changed_score"),
            "trust_delta": _delta(_mean(trust["post"]) if trust["post"] else None,
                                  _mean(trust["pre"]) if trust["pre"] else None),
            "coop_delta": _delta(_rate(coop["post"]), _rate(coop["pre"])),
            "gap_delta": _delta(_rate(gap["post"]), _rate(gap["pre"])),
        })
    return out


# ---------------------------------------------------------------------------
# Bootstrap (BCa) and permutation test
# ---------------------------------------------------------------------------

def _bca(boot: List[float], theta: float, jack: List[float], alpha: float) -> tuple:
    """Bias-corrected and accelerated percentile interval; falls back to plain percentiles."""
    boot = sorted(boot)
    n = len(boot)

    def pick(p: float) -> float:
        return boot[min(n - 1, max(0, int(p * n)))]

    below = sum(b < theta for b in boot) / n
    below = min(max(below, 1 / (n + 1)), 1 - 1 / (n + 1))
    z0 = _ND.inv_cdf(below)
    jm = _mean(jack)
    den = 6 * (sum((jm - j) ** 2 for j in jack)) ** 1.5
    acc = sum((jm - j) ** 3 for j in jack) / den if den > 1e-12 else 0.0

    def adjusted(q: float) -> float:
        z = _ND.inv_cdf(q)
        denom = 1 - acc * (z0 + z)
        if denom <= 1e-9:
            return q
        return _ND.cdf(z0 + (z0 + z) / denom)

    return pick(adjusted(alpha / 2)), pick(adjusted(1 - alpha / 2))


def bootstrap_ci(values: Sequence[Optional[float]], n_boot: int = 10000, alpha: float = 0.05,
                 seed: int = 0) -> Optional[Dict[str, Any]]:
    xs = _clean(values)
    if len(xs) < 2:
        return None
    rng = random.Random(seed)
    n = len(xs)
    boot = [_mean([xs[rng.randrange(n)] for _ in range(n)]) for _ in range(n_boot)]
    theta = _mean(xs)
    jack = [_mean(xs[:i] + xs[i + 1:]) for i in range(n)]
    lo, hi = _bca(boot, theta, jack, alpha)
    return {"mean": theta, "lo": lo, "hi": hi, "n": n}


def diff_ci(a: Sequence[Optional[float]], b: Sequence[Optional[float]], n_boot: int = 10000,
            alpha: float = 0.05, seed: int = 0) -> Optional[Dict[str, Any]]:
    """CI for mean(a) - mean(b); trials are resampled independently inside each arm."""
    xa, xb = _clean(a), _clean(b)
    if len(xa) < 2 or len(xb) < 2:
        return None
    rng = random.Random(seed)
    na, nb = len(xa), len(xb)
    boot = [
        _mean([xa[rng.randrange(na)] for _ in range(na)]) - _mean([xb[rng.randrange(nb)] for _ in range(nb)])
        for _ in range(n_boot)
    ]
    theta = _mean(xa) - _mean(xb)
    jack = ([_mean(xa[:i] + xa[i + 1:]) - _mean(xb) for i in range(na)]
            + [_mean(xa) - _mean(xb[:i] + xb[i + 1:]) for i in range(nb)])
    lo, hi = _bca(boot, theta, jack, alpha)
    return {"diff": theta, "lo": lo, "hi": hi, "n_a": na, "n_b": nb}


def permutation_pvalue(a: Sequence[Optional[float]], b: Sequence[Optional[float]],
                       n_perm: int = 10000, seed: int = 0) -> Optional[float]:
    """Two-sided permutation test on the difference in means (arm labels shuffled)."""
    xa, xb = _clean(a), _clean(b)
    if len(xa) < 2 or len(xb) < 2:
        return None
    rng = random.Random(seed)
    observed = abs(_mean(xa) - _mean(xb))
    pooled = xa + xb
    na = len(xa)
    hits = 0
    for _ in range(n_perm):
        rng.shuffle(pooled)
        if abs(_mean(pooled[:na]) - _mean(pooled[na:])) >= observed - 1e-12:
            hits += 1
    return (hits + 1) / (n_perm + 1)
```

"""Summarise swap-experiment arms: per-arm means with 95% BCa CIs and the pre-specified contrasts.

Usage:
  python experiments/analysis/swap_analysis.py \
      --arm a=outputs/experiments/diners/3agents/..._swap_a.json \
      --arm b=outputs/experiments/diners/3agents/..._swap_b.json [--arm c=... --arm d=...]

Each file is a run_endogenous.py output with metadata.swap. Contrasts: b-a, c-b, d-b (only
those whose arms were given). Reports unadjusted CIs and p-values and flags those that
survive a Bonferroni correction over 3 contrasts x 2 primary measures (alpha 0.05 / 6).
"""
import argparse
import json
import os
import sys

sys.path.insert(0, os.path.join(os.path.dirname(__file__), "..", ".."))

from src.endogenous.analysis.swap_stats import (  # noqa: E402
    bootstrap_ci, diff_ci, permutation_pvalue, trial_metrics,
)

PRIMARY = ("guess_score", "trust_delta")
SECONDARY = ("coop_delta", "gap_delta")
CONTRASTS = (("b", "a"), ("c", "b"), ("d", "b"))


def load_arm(path):
    with open(path) as f:
        data = json.load(f)
    swap = data["analysis"]["metadata"]["swap"] if "analysis" in data else data["metadata"]["swap"]
    return trial_metrics(data["rounds"], swap["agent"], swap["swap_at"])


def fmt(ci):
    return "n/a" if ci is None else f"{ci['mean']:.2f} [{ci['lo']:.2f}, {ci['hi']:.2f}] (n={ci['n']})"


def main():
    p = argparse.ArgumentParser()
    p.add_argument("--arm", action="append", required=True, help="label=path, e.g. a=run_a.json")
    p.add_argument("--n-boot", type=int, default=10000)
    args = p.parse_args()

    arms = {}
    for spec in args.arm:
        label, path = spec.split("=", 1)
        arms[label] = load_arm(path)

    print("\nPer-arm means with 95% BCa bootstrap CI (trial = unit)")
    for metric in PRIMARY + SECONDARY:
        print(f"\n  {metric}{'  (primary)' if metric in PRIMARY else ''}")
        for label, rows in arms.items():
            print(f"    arm {label}: {fmt(bootstrap_ci([r[metric] for r in rows], n_boot=args.n_boot))}")

    print("\nPre-specified contrasts (difference in means, 95% BCa CI, permutation p)")
    for hi, lo in CONTRASTS:
        if hi not in arms or lo not in arms:
            continue
        for metric in PRIMARY + SECONDARY:
            a, b = [r[metric] for r in arms[hi]], [r[metric] for r in arms[lo]]
            ci = diff_ci(a, b, n_boot=args.n_boot)
            pv = permutation_pvalue(a, b, n_perm=args.n_boot)
            if ci is None:
                print(f"  {hi}-{lo} {metric}: n/a")
                continue
            excl = ci["lo"] > 0 or ci["hi"] < 0
            bonf = "" if metric not in PRIMARY or pv is None else (
                "  survives Bonferroni" if pv < 0.05 / 6 else "  does not survive Bonferroni")
            print(f"  {hi}-{lo} {metric}: {ci['diff']:+.2f} [{ci['lo']:+.2f}, {ci['hi']:+.2f}] "
                  f"p={pv:.3f}  CI {'excludes' if excl else 'includes'} 0{bonf}")


if __name__ == "__main__":
    main()
